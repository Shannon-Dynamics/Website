// shannon-core: one kernel, two backends
use shannon_core::{Vec3, DisjointSlice};
/// Forward pass: scale + translate every vertex
pub fn affine_fwd(
a: &[Vec3],
scale: f32,
bias: Vec3,
out: DisjointSlice<Vec3>,
) {
let i = thread_idx();
out[i] = a[i] * scale + bias;
}
/// Adjoint: reverse-mode gradient
pub fn affine_bwd(
a: &[Vec3],
scale: f32,
d_out: &[Vec3],
sink: impl GradSink<Vec3>,
) {
let i = thread_idx();
sink.add(i, d_out[i] * scale);
}
